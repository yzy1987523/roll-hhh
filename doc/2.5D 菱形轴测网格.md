
# Godot 4 肥鹅健身房同款2.5D菱形网格房间建造完整实现方案

# 一、方案概述

本方案基于Godot 4引擎，纯2D实现《肥鹅健身房》同款2.5D菱形轴测网格房间建造系统，包含核心功能：菱形网格生成、鼠标拾取与高亮、家具放置/拖拽/删除、墙体与房间边框、建造/编辑模式切换，全程轻量化设计，无需3D建模，完美复刻肥鹅健身房房间视觉与交互逻辑，代码可直接复制运行，无缝适配美术资源替换。

# 二、核心原理

采用“2D精灵+正交相机+轴测坐标转换+深度排序”实现2.5D视觉效果，核心逻辑如下：

1. 坐标转换：通过数学公式将普通直角网格坐标，转换成斜45°菱形轴测坐标，保证视觉贴合肥鹅风格；
2. 相机设置：使用正交相机，无透视效果，避免3D渲染带来的性能损耗；
3. 深度遮挡：按物体屏幕Y轴坐标动态排序，实现“后层物体被前层遮挡”，避免穿模；
4. 轻量化实现：地板、墙体、家具均使用2D节点（ColorRect/Sprite2D），无需3D模型，适配手机小游戏场景。

# 三、节点结构（必按此搭建）

节点层级清晰，无需额外复杂节点，直接复制结构即可：

```plain
RoomRoot（Node2D，根节点）
 ├─ Camera2D（勾选 Current，正交相机）
 └─ GridManager（Node2D，挂载所有核心脚本）
```

补充说明：Camera2D无需旋转，仅需勾选“Current”，后期可微调位置以完整显示整个房间。

# 四、完整核心脚本（可直接复制运行）

脚本挂载在GridManager节点上，整合所有功能（网格、放置、拖拽、删除、墙体、模式切换），零外部依赖，无素材也可运行。

```gdscript
extends Node2D

# ================================= 基础配置（肥鹅标准参数） =================================
# 网格配置（2:1菱形格子，肥鹅健身房通用比例）
@export var cell_size: Vector2 = Vector2(128, 64)
@export var map_width: int = 10
@export var map_height: int = 8

# 颜色配置（无素材时用纯色块，替换美术资源后可删除）
@export var floor_color: Color = Color(0.28, 0.24, 0.20)
@export var highlight_color: Color = Color(0.3, 1.0, 0.3, 0.4)
@export var occupy_color: Color = Color(0.9, 0.3, 0.3, 0.5)

# 墙体配置
@export var wall_thickness: int = 32
@export var wall_color_back: Color = Color(0.6, 0.55, 0.5)
@export var wall_color_side: Color = Color(0.5, 0.45, 0.4)

# 删除功能配置
@export var can_delete: bool = true
@export var delete_key: Key = KEY_DELETE

# 建造/编辑模式配置
@export var is_edit_mode: bool = false
@export var edit_mode_key: Key = KEY_TAB

# ================================= 数据存储 =================================
var floor_list: Array<ColorRect> = []          # 地板节点列表
var preview_tile: ColorRect = null             # 鼠标预览格子
var grid_occupied: Array<Array<bool>> = []     # 格子占用状态（true=已占用）
var furniture_root: Node2D                     # 家具父节点（统一管理）
var wall_root: Node2D                         # 墙体父节点（统一管理）
var selected_furniture: CanvasItem = null      # 当前选中的家具

# ================================= 初始化 =================================
func _ready():
    # 1. 初始化家具父节点
    furniture_root = Node2D.new()
    furniture_root.name = "FurnitureRoot"
    add_child(furniture_root)

    # 2. 初始化墙体父节点
    wall_root = Node2D.new()
    wall_root.name = "WallRoot"
    add_child(wall_root)

    # 3. 初始化格子占用数组（默认全部未占用）
    grid_occupied = []
    for y in range(map_height):
        var row = []
        for x in range(map_width):
            row.append(false)
        grid_occupied.append(row)

    # 4. 初始化预览格子、生成地板、生成墙体
    init_preview()
    generate_isometric_grid()
    generate_room_walls()

# ================================= 核心：坐标转换（等轴测菱形） =================================
# 网格坐标 → 世界坐标（用于摆放地板、家具、墙体）
func grid_to_world(gx: int, gy: int) -> Vector2:
    var half_w = cell_size.x * 0.5
    var half_h = cell_size.y * 0.5
    var wx = (gx - gy) * half_w
    var wy = (gx + gy) * half_h
    return Vector2(wx, wy)

# 世界坐标 → 网格坐标（用于鼠标拾取、拖拽吸附）
func world_to_grid(world_pos: Vector2) -> Vector2i:
    var half_w = cell_size.x * 0.5
    var half_h = cell_size.y * 0.5
    var gx = (world_pos.x / half_w + world_pos.y / half_h) * 0.5
    var gy = (world_pos.y / half_h - world_pos.x / half_w) * 0.5
    return Vector2i(round(gx), round(gy))

# ================================= 地板生成 =================================
func generate_isometric_grid():
    # 批量生成菱形地板（无素材用纯色块，替换时改为Sprite2D即可）
    for gx in range(map_width):
        for gy in range(map_height):
            var floor = ColorRect.new()
            floor.size = cell_size
            floor.color = floor_color
            floor.position = grid_to_world(gx, gy)
            floor.pivot_offset = cell_size * 0.5  # 锚点居中，保证菱形裁切正确
            floor.rect_clip_mode = CanvasItem.RECT_CLIP_MODE.DIAMOND  # 菱形裁切
            floor_list.append(floor)
            add_child(floor)

# 初始化鼠标预览格子（高亮提示当前选中格子）
func init_preview():
    preview_tile = ColorRect.new()
    preview_tile.size = cell_size
    preview_tile.pivot_offset = cell_size * 0.5
    preview_tile.rect_clip_mode = CanvasItem.RECT_CLIP_MODE.DIAMOND
    add_child(preview_tile)

# ================================= 墙体 & 房间边框生成 =================================
func generate_room_walls():
    # 1. 后墙（地图最上沿 gy=0 整排，贴合肥鹅后墙位置）
    for gx in range(map_width):
        var wall = ColorRect.new()
        wall.size = Vector2(cell_size.x, wall_thickness)
        wall.color = wall_color_back
        wall.pivot_offset = Vector2(cell_size.x * 0.5, wall_thickness)
        var base_pos = grid_to_world(gx, 0)
        wall.position = base_pos + Vector2(0, -wall_thickness * 0.5)  # 向上偏移，贴紧地板上沿
        wall.rect_clip_mode = CanvasItem.RECT_CLIP_MODE.DIAMOND
        wall.sorting_order = 1000  # 层级置顶，不被家具遮挡
        wall_root.add_child(wall)

    # 2. 左侧墙（gx=0 整列）
    for gy in range(map_height):
        var side_wall = ColorRect.new()
        side_wall.size = Vector2(wall_thickness, cell_size.y)
        side_wall.color = wall_color_side
        side_wall.pivot_offset = Vector2(wall_thickness, cell_size.y * 0.5)
        side_wall.position = grid_to_world(0, gy) + Vector2(-wall_thickness * 0.3, 0)
        side_wall.sorting_order = 999  # 层级仅次于后墙
        wall_root.add_child(side_wall)

    # 3. 右侧墙（gx=map_width-1 整列）
    for gy in range(map_height):
        var side_wall = ColorRect.new()
        side_wall.size = Vector2(wall_thickness, cell_size.y)
        side_wall.color = wall_color_side
        side_wall.pivot_offset = Vector2(0, cell_size.y * 0.5)
        side_wall.position = grid_to_world(map_width - 1, gy) + Vector2(wall_thickness * 0.3, 0)
        side_wall.sorting_order = 999
        wall_root.add_child(side_wall)

    # 4. 边界锁定：墙体所在格子永久占用，禁止放置家具
    set_map_border_limit()

# 边界规则：后墙、左右侧墙所在格子锁定，无法放置家具
func set_map_border_limit():
    # 后墙行（gy=0）全部锁定
    for x in range(map_width):
        grid_occupied[0][x] = true
    # 左右侧墙列（gx=0、gx=map_width-1）全部锁定
    for y in range(map_height):
        grid_occupied[y][0] = true
        grid_occupied[y][map_width - 1] = true

# ================================= 每帧更新（高亮、排序） =================================
func _process(delta):
    refresh_furniture_sort()  # 2.5D遮挡排序，实时更新
    update_mouse_preview()    # 鼠标格子高亮，实时更新

# 鼠标格子高亮预览（区分可放置/已占用/非编辑模式）
func update_mouse_preview():
    var mouse_g = world_to_grid(get_global_mouse_position())
    if is_in_map(mouse_g.x, mouse_g.y):
        preview_tile.visible = true
        preview_tile.position = grid_to_world(mouse_g.x, mouse_g.y)
        # 颜色区分：已占用→红色，可放置→绿色；非编辑模式→灰化
        preview_tile.color = grid_occupied[mouse_g.y][mouse_g.x] ? occupy_color : highlight_color
        preview_tile.modulate = is_edit_mode ? Color(1,1,1) : Color(0.5,0.5,0.5)
    else:
        preview_tile.visible = false

# 判断格子是否在地图范围内
func is_in_map(x: int, y: int) -> bool:
    return x >= 0 && y >= 0 && x < map_width && y < map_height

# 2.5D核心：家具动态排序（按屏幕Y轴，Y越大越靠下，越先渲染，避免穿模）
func refresh_furniture_sort():
    for item in furniture_root.get_children():
        if item is CanvasItem:
            item.sorting_order = -int(item.global_position.y)

# ================================= 建造/编辑模式控制 =================================
# 切换编辑模式（Tab键触发，也可绑定UI按钮）
func toggle_edit_mode():
    is_edit_mode = !is_edit_mode
    # 退出编辑模式时，清空选中的家具，防止误操作
    if not is_edit_mode:
        selected_furniture = null

# 判断是否可进行操作（只有编辑模式可操作）
func can_operate() -> bool:
    return is_edit_mode

# ================================= 输入交互（放置/拖拽/删除） =================================
func _input(event: InputEvent):
    # 1. Tab键切换建造/编辑模式
    if event is InputEventKey and event.pressed and event.keycode == edit_mode_key:
        toggle_edit_mode()
        return

    # 2. 鼠标点击操作（放置/选中）
    if event is InputEventMouseButton and event.pressed:
        var mouse_g = world_to_grid(get_global_mouse_position())
        if not is_in_map(mouse_g.x, mouse_g.y):
            return
      
        # 左键：放置家具 / 选中家具
        if event.button_index == MOUSE_BUTTON_LEFT:
            if not can_operate(): return  # 非编辑模式禁止操作
          
            var click_fur = get_furniture_at_grid(mouse_g)
            if click_fur != null:
                selected_furniture = click_fur  # 点击已有家具→选中
                return
            if not grid_occupied[mouse_g.y][mouse_g.x]:
                spawn_furniture(mouse_g.x, mouse_g.y)  # 点击空地→放置家具

        # 右键：删除选中家具 / 取消选中
        if event.button_index == MOUSE_BUTTON_RIGHT:
            if not can_operate(): return  # 非编辑模式禁止操作
          
            var click_fur = get_furniture_at_grid(mouse_g)
            if click_fur != null:
                delete_furniture(click_fur)  # 右键点击家具→删除
                selected_furniture = null
            else:
                selected_furniture = null  # 右键点击空地→取消选中

    # 3. 拖拽移动选中的家具
    if event is InputEventMouseMotion and selected_furniture != null:
        if not can_operate(): return  # 非编辑模式禁止操作
      
        var mouse_g = world_to_grid(get_global_mouse_position())
        if is_in_map(mouse_g.x, mouse_g.y) and not grid_occupied[mouse_g.y][mouse_g.x]:
            clear_furniture_grid_occupy(selected_furniture)  # 清空旧格子占用
            selected_furniture.position = grid_to_world(mouse_g.x, mouse_g.y)  # 移动到新格子
            grid_occupied[mouse_g.y][mouse_g.x] = true  # 标记新格子为占用

    # 4. Delete键删除选中家具
    if event is InputEventKey and event.pressed and event.keycode == delete_key:
        if not can_operate(): return  # 非编辑模式禁止操作
        if selected_furniture != null:
            delete_furniture(selected_furniture)

# ================================= 家具操作（生成/删除） =================================
# 生成家具（在指定网格坐标放置）
func spawn_furniture(gx: int, gy: int):
    var fur = ColorRect.new()
    fur.name = "Furniture"
    fur.size = Vector2(96, 96)
    fur.color = Color(0.7, 0.55, 0.4)  # 纯色占位，替换为Sprite2D即可
    fur.pivot_offset = Vector2(fur.size.x / 2, fur.size.y)  # 锚点底部，保证踩地正确
    fur.position = grid_to_world(gx, gy)
    furniture_root.add_child(fur)
    grid_occupied[gy][gx] = true  # 标记格子为占用
    return fur

# 删除家具（同时清空格子占用）
func delete_furniture(fur: CanvasItem) -> void:
    if not can_delete:
        return
    clear_furniture_grid_occupy(fur)  # 清空格子占用
    fur.queue_free()  # 移除家具节点
    if selected_furniture == fur:
        selected_furniture = null  # 清空选中状态

# 清空家具所在格子的占用状态
func clear_furniture_grid_occupy(fur: CanvasItem):
    var old_g = world_to_grid(fur.position)
    if is_in_map(old_g.x, old_g.y):
        grid_occupied[old_g.y][old_g.x] = false

# 根据网格坐标获取家具（用于点击选中、右键删除）
func get_furniture_at_grid(target_g: Vector2i) -> CanvasItem:
    for fur in furniture_root.get_children():
        var g = world_to_grid(fur.position)
        if g == target_g:
            return fur
    return null
```

# 五、关键功能操作说明（运行即用）

## 5.1 基础操作

* 鼠标移动：预览格子跟随鼠标，绿色=可放置，红色=已占用/墙体，灰色=非编辑模式；
* Tab键：切换建造/编辑模式（默认非编辑模式，无法操作）；
* 放置家具：编辑模式下，左键点击空地，生成家具；
* 选中家具：编辑模式下，左键点击已有家具；
* 拖拽家具：选中家具后，拖动鼠标，家具吸附到格子移动（仅可移动到空格子）；
* 删除家具：编辑模式下，右键点击家具，或选中家具后按Delete键。

## 5.2 模式区分

| 模式                                                                       | 状态                         | 可操作项                   |
| -------------------------------------------------------------------------- | ---------------------------- | -------------------------- |
| -------------------------------------------------------------------------- |                              |                            |
| 普通模式（默认）                                                           | 预览格子灰化，无选中状态     | 无（禁止放置、拖拽、删除） |
| -                                                                          | -                            | -                          |
| 编辑模式（Tab切换）                                                        | 预览格子正常显色，可选中家具 | 放置、拖拽、删除家具       |

## 5.3 墙体规则

* 墙体包含：后墙（最上排）、左侧墙（最左列）、右侧墙（最右列）；
* 墙体层级置顶，永远不会被家具遮挡；
* 墙体所在格子永久锁定，无法放置家具，鼠标悬浮显示红色。

# 六、美术资源替换方案（一键切换）

当前脚本用ColorRect做占位，替换为肥鹅风格美术资源时，仅需修改以下3处，无需改动核心逻辑：

## 6.1 地板替换

找到generate_isometric_grid()函数，将ColorRect替换为Sprite2D，赋值地砖贴图：

```gdscript
# 替换前
var floor = ColorRect.new()
floor.size = cell_size
floor.color = floor_color

# 替换后
var floor = Sprite2D.new()
floor.texture = preload("res://地板贴图路径.png")  # 导入你的轴测地砖图
floor.size = cell_size  # 贴图尺寸需与cell_size一致（128x64）
```

## 6.2 家具替换

找到spawn_furniture()函数，将ColorRect替换为Sprite2D，赋值家具贴图：

```gdscript
# 替换前
var fur = ColorRect.new()
fur.size = Vector2(96, 96)
fur.color = Color(0.7, 0.55, 0.4)

# 替换后
var fur = Sprite2D.new()
fur.texture = preload("res://家具贴图路径.png")  # 导入你的轴测家具图
fur.size = Vector2(96, 96)  # 贴图尺寸可根据实际调整
```

关键注意：家具贴图需按45°轴测视角绘制，锚点保留“底部中心”（pivot_offset = Vector2(fur.size.x/2, fur.size.y)），保证踩地正确。

## 6.3 墙体替换

找到generate_room_walls()函数，将墙体的ColorRect替换为Sprite2D，分别赋值后墙、侧墙贴图：

```gdscript
# 后墙替换示例（侧墙同理）
# 替换前
var wall = ColorRect.new()
wall.size = Vector2(cell_size.x, wall_thickness)
wall.color = wall_color_back

# 替换后
var wall = Sprite2D.new()
wall.texture = preload("res://后墙贴图路径.png")
```

# 八、常见问题排查

* 家具穿模：检查家具锚点是否为底部中心，refresh_furniture_sort()函数是否正常调用；
* 格子吸附异常：检查坐标转换公式，确保cell_size为128x64（2:1比例）；
* 墙体被家具遮挡：检查墙体的sorting_order是否为999/1000，确保层级高于家具；
* 无法删除家具：检查can_delete是否为true，是否处于编辑模式。
